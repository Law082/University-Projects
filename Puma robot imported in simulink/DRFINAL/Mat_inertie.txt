function M = inertia_mat(q)

xc_1 = 0; yc_1 = -10^-3 * 24.297; zc_1 = 205.643 * 10^-3;
Ixx_1 = 1.112; Iyy_1 = 0.659; Izz_1 = 0.944; m_1 = 41.894;

xc_2 = -10^-3 * 304.716; yc_2 = -10^-3 * 64.285; zc_2 = 0;
Ixx_2 = 3.774; Iyy_2 = 0.402; Izz_2 = 3.471; m_2 = 40.215;

xc_3 = -0.002 * 10^-3; yc_3 = 29.318 * 10^-3; zc_3 = -268.867 * 10^-3;
Ixx_3 = 0.458; Iyy_3 = 0.492; Izz_3 = 0.037; m_3 = 7.842;

x_2 = 0; y_2 = -270 * 10 ^-3; z_2 = 240 * 10^-3;
x_3 = -730.827 * 10^-3 ;y_3 = 0; z_3 = 0;



m_11 = m_1 * (xc_1^2 + yc_1^2) + m_2 * ((xc_2*cos(q(2,:)) + zc_2 * sin(q(2,:)) + x_2)^2 + (yc_2 + y_2)^2) + m_3 * ((xc_3*cos(q(2,:) + q(3,:)) + zc_3 * sin(q(2,:) + q(3,:)) + x_3*cos(q(2,:)) + z_3 * sin(q(2,:)) + x_2)^2 + (yc_3 + y_3 + y_2)^2) + Izz_1 + Izz_2 + Izz_3;
m_12 = -m_2 * (yc_2 + y_2) * (-xc_2 * sin(q(2,:)) + zc_2 * cos(q(2,:))) - m_3 * (yc_3 + y_3 + y_2) * (-xc_3 * sin(q(2,:) + q(3,:)) + zc_3 * cos(q(2,:) + q(3,:)) -x_3 * sin(q(2,:)) + z_3 * cos(q(2,:)));
m_13 = - m_3 * (yc_3 + y_3 + y_2) * (-xc_3 * sin(q(2,:) + q(3,:)) + zc_3 * cos(q(2,:) + q(3,:)));

m_21 = m_12;
m_22 = m_2 * ((-xc_2 * sin(q(2,:)) + zc_2 * cos(q(2,:)))^2 + (xc_2*cos(q(2,:)) + zc_2 * sin(q(2,:)))^2) + m_3 * ((-xc_3 * sin(q(2,:) + q(3,:)) + zc_3 * cos(q(2,:) + q(3,:)) -x_3 * sin(q(2,:)) + z_3 * cos(q(2,:))) ^2 + (xc_3*cos(q(2,:) + q(3,:)) + zc_3 * sin(q(2,:) + q(3,:)) + x_3*cos(q(2,:)) + z_3 * sin(q(2,:)))^2) + (Ixx_2 + Ixx_3) * sin(q(1,:))^2 + (Iyy_2 + Iyy_3) * cos(q(1,:))^2;

m_31 = m_13 ;
m_32 = m_23;
m_33 = m_3 * (xc_3 ^ 2 + zc_3 ^2) + Ixx_3 * sin(q(1,:))^2 + Iyy_3 * cos(q(1,:))^2;

M=[m_11, m_12, m_13;
    m_21, m_22, m_23;
    m_31, m_32, m_33;];
